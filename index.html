async function sendQueryToBackend(studentQuery) { const trimmedQuery = studentQuery.trim(); if (!trimmedQuery) return; const sendBtn = document.querySelector('.send'); const inputField = document.getElementById('text-input'); if (sendBtn) sendBtn.disabled = true; if (inputField) inputField.disabled = true; statusEl.innerText = "屈校長思考緊回答，請稍候..."; clearAudioBadge(); if (currentAudio) { currentAudio.pause(); currentAudio.currentTime = 0; } playWaitingAudio(); chatHistory.push({ role: "user", content: trimmedQuery }); try { const chatRes = await fetch('/api/chat', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ messages: chatHistory }) }); const chatData = await chatRes.json(); if (!chatRes.ok || !chatData.text) { stopWaitingAudio(); statusEl.innerText = `伺服器錯誤: ${chatData.detail || '無法取得回應'}`; chatHistory.pop(); return; } const displayText = chatData.text; // 繁體書面語（用於顯示） const spokenText = chatData.speech; // 廣東話口語（用於語音） let audioUrl = null; try { const ttsRes = await fetch('/api/tts', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ text: spokenText }) // 將廣東話口語傳給 TTS }); const ttsData = await ttsRes.json(); audioUrl = ttsData.audio_url; } catch (e) { console.error("TTS Fetch Error:", e); } stopWaitingAudio(); if (audioUrl) { playAudio(audioUrl, displayText, trimmedQuery, () => { chatHistory.push({ role: "assistant", content: displayText }); }); } else { statusEl.innerText = displayText; addQuestionToHistory(trimmedQuery, displayText); chatHistory.push({ role: "assistant", content: displayText }); } } catch (error) { stopWaitingAudio(); console.error("Fetch error:", error); statusEl.innerText = "請求失敗：" + error.message; clearAudioBadge(); chatHistory.pop(); } finally { if (sendBtn) sendBtn.disabled = false; if (inputField) { inputField.disabled = false; inputField.focus(); } } }
